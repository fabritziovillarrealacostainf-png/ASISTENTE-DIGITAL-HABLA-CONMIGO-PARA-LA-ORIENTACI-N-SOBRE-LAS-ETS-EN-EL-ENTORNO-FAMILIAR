        : `Hola ${context.nombrePadre}. Soy Fabrix. Te ayudaré con <b>${context.tema}</b> usando información del MINSA y estándares de la OPS/OMS. ¿Cuál es tu duda?`;
      
      addMessage("bot", welcomeMessage);
      
      // Agregar nota informativa
      setTimeout(() => {
        addMessage("bot", `<small style="opacity: 0.8;">📋 <i>Nota: Mis respuestas se basan en lineamientos oficiales de salud. Para casos específicos, consulta con un profesional de salud o llama a la Línea 113 del MINSA.</i></small>`);
      }, 1000);
    });

    sendBtn.addEventListener("click", sendMessage);
    userInput.addEventListener("keypress", e => {
      if (e.key === "Enter" && !e.shiftKey) {
        e.preventDefault();
        sendMessage();
      }
    });
